const CSAT_FOLLOWUP_DAYS = 7; // ASSUMED: days after "Survey sent" with no "Answered" date before Today asks you to chase it
function normalizeCsatBlock(ap){
  const a = ap || {};
  return Object.assign({}, a, {
    done: !!a.done, date: isRealISODate(a.date) ? a.date : null, contactName: typeof a.contactName === 'string' ? a.contactName : '',
    surveySentOn: isRealISODate(a.surveySentOn) ? a.surveySentOn : null,
    surveyAnsweredOn: isRealISODate(a.surveyAnsweredOn) ? a.surveyAnsweredOn : null});
}
function csatSurveyOverdue(c){
  const ap = c && c.compliance && c.compliance.associatedPeopleCSAT;
  if(!ap || !isRealISODate(ap.surveySentOn) || isRealISODate(ap.surveyAnsweredOn)) return null;
  const days = daysSince(ap.surveySentOn);
  return days >= CSAT_FOLLOWUP_DAYS ? {sentOn: ap.surveySentOn, days} : null;
}
function pushCsatFollowUp(c, out){
  const o = csatSurveyOverdue(c); if(!o) return;
  const d = TODAY_SOURCE_MAP.csat_unanswered;
  out.push({source:'csat_unanswered', severity:d.severity,
    headline: `CSAT survey sent ${fmtDate(o.sentOn)} and not answered`,
    reasons: [`Sent ${o.days} days ago, no answer date recorded. Chase it, then add the answer date.`],
    panelTab:'overview', chipLabel:'CSAT follow-up', why:d.why});
}
