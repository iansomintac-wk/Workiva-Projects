```js
/* D2 prep: the closing stretch of a transcript for the call-prep panel. The last n speech lines
   and everything after the first of them (speaker lines kept). Speech line = not blank, not a
   "Name  hh:mm:ss" header, not a "-->" cue, not a bare timestamp, not WEBVTT. n or fewer: whole. */
function closingStretch(text, n){
  if(typeof text !== 'string' || !text.trim()) return '';
  const keep = n > 0 ? n : 30;
  const lines = text.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
  const isHead = l => /^\S.*\s{2,}\d{1,2}:\d{2}(:\d{2})?$/.test(l) || l.includes('-->') || /^\d{1,2}:\d{2}(:\d{2})?([.,]\d+)?$/.test(l) || l === 'WEBVTT';
  const at = [];
  lines.forEach((l, i) => { if(!isHead(l)) at.push(i); });
  if(at.length <= keep) return text.trim();
  return lines.slice(at[at.length - keep]).join('\n');
}
```

/* D2 prep: last call's closing stretch on the call-prep panel. Same eligibility and order as "Last touch"
   (logged, not future-dated, newest first, ties keep array order). Hidden when no such activity has a transcript. */
function beatTranscriptSection(c){
  const acts = (DATA.activities[c.id] || []).filter(a => a && isLoggedPastDate(a.date))
    .slice().sort((a, b) => a.date < b.date ? 1 : a.date > b.date ? -1 : 0);

  for(const a of acts){ const t = activityTranscript(a); if(t && typeof t.text === 'string' && t.text.trim()){ hit = a; tr = t; break; } }

  return `
      <div class="bp-sec" data-beat-transcript-sec>
        <div class="lbl">${label}</div>
        <pre class="bp-transcript" data-beat-transcript>${escapeHtml(closingStretch(tr.text, 30))}</pre>
        <p class="bp-basis"><button class="link-btn" data-open-cust="${escapeAttr(c.id)}" data-open-tab="activity">Read the full transcript</button></p>
