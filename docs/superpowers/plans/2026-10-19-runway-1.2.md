### Task 0: Line endings (Group A; changes no html content)

Root cause (measured: section Baselines): no `.gitattributes`, `core.autocrlf=true`; both trackers are `i/lf w/crlf`, so working-tree sha256 differs from the pins (HOSTING.md and C8 1.1 pin the LF hash `7a296880...`; C8 1.0 pins the CRLF hash `a21ca8c3...`; the 1.0 LF hash is `88ff9926fd49c837103b9b0eb9ebe554f8e5cbe2a48a0b6a55926441b67a64c4`).

Decision (lead, overrides Codex finding 14): both trackers go in `.gitattributes` (`eol=lf`), the 1.0 working file is refreshed, and the C8 1.0 pin moves to its LF blob hash. Reason: the 1.0 freeze concerns its committed content, which stays byte-identical (C8's `git diff` against master guards that, and test 5 below compares the working file to the committed blob); CRLF in the working copy is a checkout artifact, and scoping the fix to 1.1 would leave the 1.0 pin dependent on whichever line endings a checkout happens to have.

// Re-pin the 1.1 tracker sha256 in docs/HOSTING.md and tools/sim/osm-day/verify.mjs (C8) from the working file.
import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { resolve } from 'node:path';
import { fileURLToPath } from 'node:url';

export function repinTexts(bytes, hostingText, verifyText) {
  if (bytes.includes(13)) throw new Error('repin: the working file has CR bytes; refresh it (rm + git restore) first');
  const sha = createHash('sha256').update(bytes).digest('hex');
  const sub = (name, text, re) => {
    if (!re.test(text)) throw new Error('repin: pattern not found in ' + name);
    return text.replace(re, (m, a, b) => a + sha + b);
  };
  return {
    sha,
    hosting: sub('docs/HOSTING.md', hostingText, /(Its sha256 is `)[0-9a-f]{64}(`)/),
    verify: sub('tools/sim/osm-day/verify.mjs', verifyText, /('1\.1': ')[0-9a-f]{64}(')/)
  };
}

if (process.argv[1] && resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  try {
    const r = repinTexts(readFileSync('osm-book-of-business-tracker-1.1.html'), readFileSync('docs/HOSTING.md', 'utf8'), readFileSync('tools/sim/osm-day/verify.mjs', 'utf8'));
    writeFileSync('docs/HOSTING.md', r.hosting); writeFileSync('tools/sim/osm-day/verify.mjs', r.verify);
    console.log('repinned 1.1 ' + r.sha.slice(0, 8));
  } catch (e) { console.error(e.message); process.exit(1); }
}

function applyCsatImportPlan(plan, customers, tickedRows){
  const byId = {}; (customers || []).forEach(c => { byId[c.id] = c; });
  const ticked = new Set(tickedRows || []);
  const written = new Set(); let fields = 0, idsWritten = 0;
  const setField = (c, ch) => {
    c.compliance = c.compliance || {};
    const ap = c.compliance.associatedPeopleCSAT = c.compliance.associatedPeopleCSAT || {done:false, date:null, contactName:'', surveySentOn:null, surveyAnsweredOn:null};
    ap[ch.field] = ch.to || null; written.add(c.id); fields++;
  };
  plan.changes.forEach(ch => { if(byId[ch.customerId]) setField(byId[ch.customerId], ch); });
  plan.suggestions.forEach(s => {
    if(!ticked.has(s.row) || !byId[s.customerId]) return;
    const c = byId[s.customerId];
    s.changes.forEach(ch => setField(c, ch));
    c.salesforceId = s.sf; idsWritten++; written.add(c.id);
    if(!String(c.gainsightId || '').trim() && s.gs){ c.gainsightId = s.gs; idsWritten++; }
  });
  return {accounts:written.size, fields:fields, idsWritten:idsWritten};

const CSAT_FOLDER_KEY = 'csat-folder';
async function csatNewestFile(dir){
  let best = null;
  for await (const h of dir.values()){
    if(!h || h.kind !== 'file' || !CSAT_FILE_RE.test(h.name)) continue;
    const f = await h.getFile();
    if(!best || f.lastModified > best.lastModified) best = {name:h.name, lastModified:f.lastModified, file:f};
  }
  return best;
}
async function csatFolderCheck(dir, request){
  if(!dir || typeof dir.queryPermission !== 'function') return {state:'none', newest:null, isNew:false};
  let st = await dir.queryPermission({mode:'read'});
  if(st === 'prompt' && request && typeof dir.requestPermission === 'function') st = await dir.requestPermission({mode:'read'});
  if(st === 'prompt') return {state:'prompt', newest:null, isNew:false};
  if(st !== 'granted') return {state:'denied', newest:null, isNew:false};
  const best = await csatNewestFile(dir);
  if(!best) return {state:'granted', newest:null, isNew:false};
  const last = (DATA.csatImport && DATA.csatImport.fileModified) || 0;
  return {state:'granted', newest:{name:best.name, lastModified:best.lastModified}, file:best.file, isNew:best.lastModified > last};
}

// Top-level argument count of the call whose "(" is at index `open` (-1 if the parentheses never balance).
function callArgs(text, open) {
  let depth = 0, n = 1, quote = null;
  for (let i = open; i < text.length; i++) {
    const ch = text[i];
    if (quote) { if (ch === '\\') { i++; continue; } if (ch === quote) quote = null; continue; }
    if (ch === "'" || ch === '"' || ch === '`') { quote = ch; continue; }
    if ('([{'.includes(ch)) depth++;
    else if (')]}'.includes(ch)) { depth--; if (depth === 0) return n; }
    else if (ch === ',' && depth === 1) n++;
  }
  return -1;
}
