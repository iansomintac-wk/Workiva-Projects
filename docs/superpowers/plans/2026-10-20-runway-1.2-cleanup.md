async function csatWatchRefresh(request){
  const w = _csatWatch;                       // 1e: remember which watch this scan belongs to
  const maxBytes = 8 * 1024 * 1024;           // 1b: same 8 MB limit as the manual import
  const done = o => { if(_csatWatch !== w) return false; Object.assign(w, o, {manual:!!request}); return true; };
  try{

    if(_csatWatch !== w) return;
    if(r.state === 'prompt') done({state:'prompt', plan:null, note:''});
