  - `saveTranscriptVerified(activityId, text): Promise<{ref, chars, sha1, truncated:false} | null>`: `null` when IndexedDB is unavailable. It first reads any existing record under the same ref; if one exists and its text matches the SHA-1 it is REUSED (no write). Otherwise it writes the record, reads it back and checks `sha1Hex(readBack.text) === sha1`. The returned object carries a NON-enumerable property `created` (true only when no record existed before this call), so it never reaches the book or JSON. On any write/read-back failure it deletes the record ONLY when no record existed before the call (it never deletes a record that existed before, which may be another tab's or the one the book points to) and throws. On success it sets the map entry and bumps `_tsVersion`.

```js
/* ============ 1.1 Step 1: TRANSCRIPT STORE ============
   Transcripts live in IndexedDB (obm-transcripts-v1), one record per transcript VERSION, keyed
   <activityId>:<12 hex of sha1>. The book keeps only {ref, chars, sha1, truncated}.
   RULE: a call never deletes a record it did not create. */
const TS_DB = 'obm-transcripts-v1';
let _tsForceDown = false, _tsDbP = null, _tsVersion = 0;
const _transcriptMap = new Map();
function tsOpen(){
  if(_tsForceDown) return Promise.resolve(null);
  if(_tsDbP) return _tsDbP;
  _tsDbP = new Promise(resolve => {
    let req;
    try{ req = indexedDB.open(TS_DB, 1); }catch(e){ resolve(null); return; }
    req.onupgradeneeded = () => {
      const db = req.result;
      db.createObjectStore('transcripts', {keyPath: 'ref'});
      db.createObjectStore('premigration', {keyPath: 'date'});
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = req.onblocked = () => resolve(null);
  });
  return _tsDbP;
}
async function tsTx(store, mode, fn){
  const db = await tsOpen();
  if(!db) throw new Error('IndexedDB unavailable');
  return new Promise((resolve, reject) => {
    const tx = db.transaction(store, mode);
    let out;
    try{ out = fn(tx.objectStore(store)); }catch(e){ reject(e); return; }
    tx.oncomplete = () => resolve(out && 'result' in out ? out.result : undefined);
    tx.onerror = tx.onabort = () => reject(tx.error || new Error('IndexedDB transaction failed'));
  });
}
const tsPut = rec => tsTx('transcripts', 'readwrite', s => s.put(rec));
function tsGet(ref){ return tsTx('transcripts', 'readonly', s => s.get(ref)); }   // a function declaration: the probe swaps window.tsGet
const tsDelete = ref => tsTx('transcripts', 'readwrite', s => s.delete(ref));
async function tsGetMany(refs){
  const out = new Map();
  if(!refs.length) return out;
  const db = await tsOpen();
  if(!db) throw new Error('IndexedDB unavailable');
  return new Promise((resolve, reject) => {
    const tx = db.transaction('transcripts', 'readonly'), s = tx.objectStore('transcripts');
    refs.forEach(r => { const q = s.get(r); q.onsuccess = () => { if(q.result) out.set(r, q.result); }; });
    tx.oncomplete = () => resolve(out);
    tx.onerror = tx.onabort = () => reject(tx.error || new Error('IndexedDB transaction failed'));
  });
}
async function tsClearAll(){ await tsTx('transcripts', 'readwrite', s => s.clear()); await tsTx('premigration', 'readwrite', s => s.clear()); }
async function sha1Hex(text){
  const buf = await crypto.subtle.digest('SHA-1', new TextEncoder().encode(text));
  return Array.from(new Uint8Array(buf), b => b.toString(16).padStart(2, '0')).join('');
```
